```js
const path = require('path')
const fs   = require('fs')

// app.getPath('music') devuelve la carpeta Música del usuario actual
const MUSIC_DIR = app.getPath('music')

function createWindow() {
  const win = new BrowserWindow({
    width: 1080, height: 700,
    minWidth: 720, minHeight: 520,
    frame: false,
    backgroundColor: '#0A0A0A',
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
    }
  })
  win.loadFile('index.html')
}

app.whenReady().then(() => {
  createWindow()
  createShortcut()
  watchMusicDir()
})
app.on('window-all-closed', () => app.quit())

// ── Acceso directo en escritorio ──────────────────────────────────────────────
function createShortcut() {
  try {
    const desktop  = app.getPath('desktop')
    const shortcut = path.join(desktop, 'Music Player.lnk')
    if (!fs.existsSync(shortcut)) {
      shell.writeShortcutLink(shortcut, {
        target:      process.execPath,
        args:        `"${app.getAppPath()}"`,
        description: 'Music Player',
        iconPath:    process.execPath,
        iconIndex:   0,
      })
    }
  } catch (e) {
    console.error('createShortcut error:', e.message)
  }
}

// ── Watcher: recarga cuando se añade música ───────────────────────────────────
function watchMusicDir() {
  let debounce = null
  try {
    fs.watch(MUSIC_DIR, { recursive: true }, (_, filename) => {
      if (!filename?.toLowerCase().endsWith('.mp3')) return
      clearTimeout(debounce)
      debounce = setTimeout(() => {
        const win = BrowserWindow.getAllWindows()[0]
        if (win) win.webContents.send('library-changed')
      }, 3000)
    })
  } catch (e) {
    console.error('watchMusicDir error:', e.message)
  }
}

// ── IPC: devolver directorio de música al renderer ────────────────────────────
ipcMain.handle('get-music-dir', () => MUSIC_DIR)

// ── IPC: escanear directorio de música (async, sin parámetro externo) ─────────
ipcMain.handle('scan-music', async () => {
  const results = []
  async function walk(d) {
    if (!fs.existsSync(d)) return
    try {
      const entries = await fs.promises.readdir(d, { withFileTypes: true })
      for (const entry of entries) {
        const full = path.join(d, entry.name)
        if (entry.isDirectory()) await walk(full)
        else if (entry.name.toLowerCase().endsWith('.mp3')) results.push(full)
      }
    } catch (e) {
      console.error('scan-music walk error:', d, e.message)
    }
  }
  await walk(MUSIC_DIR)
  results.sort()
  return results
})
```

## Task 6: Optimizar spectrum y rafLoop — detener cuando audio está pausado

**Problema:** `drawSpectrum()` (renderer.js:80) y `rafLoop()` (renderer.js:200) llaman `requestAnimationFrame` en cada frame **incluso cuando el audio está pausado**, consumiendo CPU/batería innecesariamente.

**Archivos:**
- Modify: `renderer.js` — funciones `drawSpectrum`, `rafLoop`, y los event listeners de `audio`

- [ ] **Step 1: Reescribir `drawSpectrum` para detener el loop cuando pausa**

Reemplazar la función `drawSpectrum` y su llamada inicial (renderer.js líneas 79-124):

```js
// ── Spectrum ──────────────────────────────────────────────────────────────────
const canvas   = document.getElementById('spectrum')
const ctx2d    = canvas.getContext('2d')
const fftData  = new Uint8Array(analyser.frequencyBinCount)
const smoothed = new Float32Array(BAR_COUNT)

let _cW = 0, _cH = 0
let spectrumRafId = null

function drawSpectrum() {
  const W = canvas.offsetWidth
  const H = canvas.offsetHeight
  if (_cW !== W || _cH !== H) {
    canvas.width = _cW = W
    canvas.height = _cH = H
  }
  ctx2d.clearRect(0, 0, W, H)

  if (audio.paused) {
    spectrumRafId = null
    return  // detener loop — se reinicia desde el evento 'play'
  }

  spectrumRafId = requestAnimationFrame(drawSpectrum)

  analyser.getByteFrequencyData(fftData)

  const grad = ctx2d.createLinearGradient(0, H, 0, 0)
  grad.addColorStop(0,    spectrumColors.dark)
  grad.addColorStop(0.45, spectrumColors.mid)
  grad.addColorStop(1,    spectrumColors.light)
  ctx2d.fillStyle = grad

  const gap = 2
  const bw  = (W - gap * (BAR_COUNT - 1)) / BAR_COUNT
  const len = fftData.length

  for (let i = 0; i < BAR_COUNT; i++) {
    const lo = Math.floor(Math.pow(len * 0.8, i / BAR_COUNT))
    const hi = Math.ceil(Math.pow(len * 0.8, (i + 1) / BAR_COUNT))
    let peak = 0
    for (let j = lo; j < Math.min(hi, len); j++)
      if (fftData[j] > peak) peak = fftData[j]

    const target = peak / 255
    smoothed[i] = smoothed[i] > target
      ? smoothed[i] * 0.82
      : smoothed[i] * 0.25 + target * 0.75

    const bh = Math.max(smoothed[i] * H, 2)
    const x  = i * (bw + gap)
    ctx2d.beginPath()
    ctx2d.roundRect(x, H - bh, bw, bh, 2)
    ctx2d.fill()
  }
}
// No llamar drawSpectrum() aquí — se inicia desde el evento 'play'
```

## Task 7: Corregir parseo de timestamps LRC

**Problema:** `renderer.js:499-500` parsea la fracción de segundos de forma frágil. El enfoque `parseFloat` es más robusto para cualquier precisión (1, 2 o 3 dígitos).

**Archivos:**
- Modify: `renderer.js` función `parseLrc` (líneas 499-501)

- [ ] **Step 1: Actualizar la línea de cálculo de ms en `parseLrc`**

Reemplazar líneas 499-501 dentro de `parseLrc`:
```js
// ANTES:
const frac = m[3]
const ms   = parseInt(frac) * (frac.length === 2 ? 10 : 1)
const time = parseInt(m[1]) * 60 + parseInt(m[2]) + ms / 1000

// DESPUÉS:
const ms   = Math.round(parseFloat('0.' + m[3]) * 1000)

```

`parseFloat('0.12') * 1000` = 120ms, `parseFloat('0.123') * 1000` = 123ms, `parseFloat('0.1') * 1000` = 100ms — funciona para 1, 2 y 3 dígitos sin lógica condicional.
